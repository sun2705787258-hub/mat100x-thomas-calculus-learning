\section*{Chapter 6 - Applications of Definite Integrals}

% Volume by slicing; textbook page 333, PDF page 334
\[
  V=\int_a^b A(x)\,dx.
\]

% Disk and washer methods; textbook pages 335 and 338, PDF pages 336 and 339
\[
  V=\pi\int_a^b\left[R(x)\right]^2dx.
\]
\[
  V=\pi\int_a^b\left(\left[R(x)\right]^2-\left[r(x)\right]^2\right)dx.
\]

% Shell method; textbook page 346, PDF page 347
\[
  V=\int_a^b 2\pi(\text{shell radius})(\text{shell height})\,dx.
\]

% Arc length; textbook page 352, PDF page 353
\[
  L=\int_a^b\sqrt{1+\left(\frac{dy}{dx}\right)^2}\,dx.
\]
\[
  ds=\sqrt{dx^2+dy^2}.
\]

% Surface area of revolution; textbook page 358, PDF page 359
\[
  S=\int_a^b 2\pi f(x)\sqrt{1+\left(f'(x)\right)^2}\,dx.
\]

% Work by variable force and Hooke's Law; textbook page 363, PDF page 364
\[
  W=\int_a^b F(x)\,dx,
  \qquad
  F(x)=kx.
\]

% Fluid force on a vertical plate; textbook page 366, PDF page 367
\[
  F=\int_a^b w\,(\text{strip depth})L(y)\,dy.
\]

% Mass, moments, centroid; textbook pages 373-376, PDF pages 374-377
\[
  M=\int_a^b\delta(x)\,dx,
  \qquad
  \bar{x}=\frac{M_y}{M},
  \qquad
  \bar{y}=\frac{M_x}{M}.
\]

% Pappus theorems; textbook pages 379 and 381, PDF pages 380 and 382
\[
  V=2\pi rA,
\]
\[
  S=2\pi rL.
\]
